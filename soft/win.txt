function Get-SystemInfo {
    try {
        $os = (Get-CimInstance Win32_OperatingSystem -ErrorAction SilentlyContinue).Caption
        $cpu = (Get-CimInstance Win32_Processor -ErrorAction SilentlyContinue).Name
        $ram = [math]::Round((Get-CimInstance Win32_PhysicalMemory -ErrorAction SilentlyContinue | Measure-Object Capacity -Sum).Sum / 1GB)
        $lblH1.Text = "HĐH: $os | CPU: $cpu | RAM: $ram GB"
    } catch { 
        $lblH1.Text = "Không thể tải cấu hình thông tin hệ thống." 
    }
}

# TỐI ƯU: Sử dụng .NET Registry để cải thiện tốc độ quét danh sách app (không gây treo Form)
function Scan-Apps {
    $listUn.Items.Clear()
    $global:UnTable = @{}
    
    $regPaths = @(
        "SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall",
        "SOFTWARE\Wow6432Node\Microsoft\Windows\CurrentVersion\Uninstall"
    )
    
    # Quét HKLM và HKCU
    $hives = @([Microsoft.Win32.Registry]::LocalMachine, [Microsoft.Win32.Registry]::CurrentUser)
    
    foreach ($hive in $hives) {
        foreach ($rp in $regPaths) {
            $key = $hive.OpenSubKey($rp)
            if ($key) {
                foreach ($subKeyName in $key.GetSubKeyNames()) {
                    $subKey = $key.OpenSubKey($subKeyName)
                    if ($subKey) {
                        $displayName = $subKey.GetValue("DisplayName")
                        $uninstallString = $subKey.GetValue("UninstallString")
                        $systemComponent = $subKey.GetValue("SystemComponent")
                        $releaseType = $subKey.GetValue("ReleaseType")
                        
                        if ($displayName -and $uninstallString -and ($systemComponent -ne 1) -and ($releaseType -ne "Update")) {
                            if ($displayName -notmatch "Microsoft\.NET" -and $displayName -notmatch "Visual C\+\+") {
                                $name = $displayName.ToString().Trim()
                                if (!$global:UnTable.ContainsKey($name)) {
                                    $global:UnTable[$name] = @{ Type="Win32"; Un=$uninstallString.ToString() }
                                    [void]$listUn.Items.Add($name)
                                }
                            }
                        }
                        $subKey.Close()
                    }
                }
                $key.Close()
            }
        }
    }
}

# TỐI ƯU: Loại bỏ hoàn toàn Get-ChildItem -Recurse, định vị trực tiếp đường dẫn cố định
function Create-DesktopShortcut ($itemName) {
    try {
        $wshShell = New-Object -ComObject WScript.Shell
        $desktopPath = [System.IO.Path]::Combine([Environment]::GetFolderPath("Desktop"), "$itemName.lnk")
        
        # Ánh xạ trực tiếp đường dẫn mặc định của các phần mềm phổ biến
        $targetExe = switch -regex ($itemName) {
            "Chrome"      { "C:\Program Files\Google\Chrome\Application\chrome.exe" }
            "Edge"        { "C:\Program Files (x86)\Microsoft\Edge\Application\msedge.exe" }
            "Cốc Cốc"     { "$env:LocalAppData\CocCoc\Browser\Application\browser.exe" }
            "Firefox"     { "C:\Program Files\Mozilla Firefox\firefox.exe" }
            "AnyDesk"     { "C:\Program Files (x86)\AnyDesk\AnyDesk.exe" }
            "UltraViewer" { "C:\Program Files (x86)\UltraViewer\UltraViewer_Desktop.exe" }
            "7-Zip"       { "C:\Program Files\7-Zip\7zFM.exe" }
            "WinRAR"      { "C:\Program Files\WinRAR\WinRAR.exe" }
            "UniKey"      { "C:\Program Files\UniKey\UniKeyNT.exe" }
            "EVKey"       { if (Test-Path "C:\Program Files\EVKey\EVKey64.exe") { "C:\Program Files\EVKey\EVKey64.exe" } else { "C:\Program Files\EVKey\EVKey32.exe" } }
            "VLC"         { "C:\Program Files\VideoLAN\VLC\vlc.exe" }
            "Zalo"        { "$env:ProgramFiles\Zalo PC\Zalo.exe" }
            default       { $null }
        }
        
        if ($targetExe -and (Test-Path $targetExe)) {
            $shortcut = $wshShell.CreateShortcut($desktopPath)
            $shortcut.TargetPath = $targetExe
            $shortcut.WorkingDirectory = Split-Path $targetExe
            $shortcut.Save()
            Add-Log "Đã tạo lối tắt ngoài Desktop: $itemName"
        }
    } catch { 
        Add-Log "Không thể tạo shortcut Desktop cho $itemName" 
    } finally {
        if ($wshShell) { [System.Runtime.InteropServices.Marshal]::ReleaseComObject($wshShell) | Out-Null }
    }
}

    $sel = $listTw.CheckedItems; if($sel.Count -eq 0){return}
    $c = 0; $t = $sel.Count
    foreach($i in $sel){
        $c++; Add-Log "Đang xử lý: $i"; Update-P $c $t
        try {
            $needExplorerRestart = $false
            if ($i -match "01. Gỡ bỏ") {
                 $WhiteList = "WindowsStore|Calculator|Photos|Paint|Notepad|StickyNotes|Framework|Runtime|VCLibs|UI.Xaml|NET.Native"
                 Get-AppxPackage -AllUsers | ForEach-Object {
                      if ($_.Name -notmatch $WhiteList -and $_.NonRemovable -ne $true) {
                           $PkgName = $_.Name
                           Remove-AppxPackage -Package $_.PackageFullName -AllUsers -ErrorAction SilentlyContinue
                           Get-AppxProvisionedPackage -Online | Where-Object { $_.DisplayName -eq $PkgName -or $_.PackageName -match $PkgName } | Remove-AppxProvisionedPackage -Online -ErrorAction SilentlyContinue } } }
            if ($i -match "02. Dọn dẹp") {
                 Remove-Item "$env:TEMP\*", "C:\Windows\Temp\*" -Force -Recurse -ea 0
                 [void](Clear-RecycleBin -Force -Confirm:$false -ea 0) }
            if ($i -match "03. Tắt ứng dụng") {
                 $wl = "EVKey|UniKey|GoogleDriveFS|SecurityHealth"
                 $paths = "HKCU:\Software\Microsoft\Windows\CurrentVersion\Run", "HKLM:\Software\Microsoft\Windows\CurrentVersion\Run"
                 foreach ($path in $paths) {
                      if (Test-Path $path) {
                           (Get-Item $path -ea 0).Property | ForEach-Object {

# TỐI ƯU: Đổi từ Đệ quy sang Vòng lặp Whilte lướt Queue tuần tự (An toàn cho bộ nhớ)
function Start-InstallationQueue {
    if (-not (Get-Command winget -ErrorAction SilentlyContinue)) {
        Add-Log "Lỗi: Không tìm thấy Winget. Hãy cập nhật App Installer."
        [System.Windows.Forms.MessageBox]::Show("Hệ thống thiếu công cụ Winget của Microsoft.", "Cảnh báo", "OK", "Warning")
        return
    }

    $total = $global:DownloadQueue.Count
    while ($global:CurrentDownloadIndex -lt $total) {
        $task = $global:DownloadQueue[$global:CurrentDownloadIndex]
        $item = $task.Item
        $id = $task.Id

        Add-Log "Đang xử lý: $item"
        $pos = [math]::Round((($global:CurrentDownloadIndex + 0.5) / $total) * 100)
        Update-P $pos 100
        
        Add-Log "Đang chạy cài đặt ngầm: $id..."
        $proc = Start-Process winget -ArgumentList "install --id $id --silent --force --accept-package-agreements --accept-source-agreements" -Wait -NoNewWindow -PassThru
        
        if ($proc.ExitCode -eq 0) {
            Add-Log "Cài đặt thành công: $item"
        } else { 
            Add-Log "Hoàn tất kiểm tra: $item (Mã: $($proc.ExitCode))" 
        }
        Create-DesktopShortcut $item
        $global:CurrentDownloadIndex++
    }
    Update-P 100 100
    Add-Log "Cài đặt hoàn tất tất cả phần mềm được chọn!"
    [System.Windows.Forms.MessageBox]::Show("Quá trình cài đặt tự động hoàn tất!", "Thông báo", "OK", "Information")
}
