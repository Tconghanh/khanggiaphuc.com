function Write-Log {
    param($Msg,[ValidateSet("OK","WARN","ERR","INFO","TITLE")]$Type="INFO")
    $ts=$( (Get-Date).ToString("HH:mm:ss") )
    $icon=switch($Type){"OK"{"[OK]"};"WARN"{"[!!]"};"ERR"{"[XX]"};"INFO"{"[>>]"};"TITLE"{"[==]"}}
    $line="[$ts] $icon $Msg"
    $color=switch($Type){"OK"{$C.LogGreen};"WARN"{$C.LogYellow};"ERR"{$C.LogRed};"INFO"{$C.LogCyan};"TITLE"{$C.Accent}}
    $txtLog.SelectionStart=$txtLog.TextLength; $txtLog.SelectionLength=0
    $txtLog.SelectionColor=$color; $txtLog.AppendText("$line`n"); $txtLog.ScrollToCaret()
    $global:LogLines.Add($line)
    [System.Windows.Forms.Application]::DoEvents()
}
function Set-Progress {
    param([int]$Val,[int]$Max=100)
    $p=[math]::Min(100,[math]::Max(0,[math]::Round($Val*100/$Max)))
    $progressBar.Value=$p; $lblPct.Text="$p%"
    [System.Windows.Forms.Application]::DoEvents()
}
function Reset-Progress { $progressBar.Value=0; $lblPct.Text="0%"; [System.Windows.Forms.Application]::DoEvents() }

$btnRunTw.Add_Click({
    $sel=$listTw.CheckedItems
    if ($sel.Count -eq 0) { Write-Log "Chưa chọn mục nào." "WARN"; return }
    Reset-Progress
    Write-Log "BẮT ĐẦU TÙY CHỈNH HỆ THỐNG ($($sel.Count) mục)" "TITLE"
    $c=0; $needRestart=$false; $needExplorer=$false

    foreach ($item in $sel) {
        $c++; Set-Progress $c $sel.Count
        Write-Log "Đang xử lý: ${item}" "INFO"
        try {
            if ($item -match "^01") {
                $wl="WindowsStore|Calculator|Photos|Paint|Notepad|StickyNotes|Framework|Runtime|VCLibs|UI\.Xaml|NET\.Native|ScreenSketch|Snip"
                Get-AppxPackage -AllUsers | Where-Object { $_.Name -notmatch $wl -and $_.NonRemovable -ne $true } | ForEach-Object {
                    Remove-AppxPackage -Package $_.PackageFullName -AllUsers -ErrorAction SilentlyContinue
                }
                Write-Log "Gỡ bloatware hoàn tất." "OK"
            }
            if ($item -match "^02") {
                Remove-Item "$env:TEMP\*"           -Force -Recurse -ErrorAction SilentlyContinue
                Remove-Item "C:\Windows\Temp\*"   -Force -Recurse -ErrorAction SilentlyContinue
                Remove-Item "C:\Windows\Prefetch\*" -Force -Recurse -ErrorAction SilentlyContinue
                [void](Clear-RecycleBin -Force -Confirm:$false -ErrorAction SilentlyContinue)
                Start-Process "dism.exe" -ArgumentList "/Online /Cleanup-Image /StartComponentCleanup" -WindowStyle Hidden -Wait
                Write-Log "Dọn ổ đĩa hoàn tất." "OK"
            }
            if ($item -match "^03") {
                Remove-Item "C:\Windows\Prefetch\*" -Force -Recurse -ErrorAction SilentlyContinue
                Stop-Service "SysMain" -Force -ErrorAction SilentlyContinue
                Set-Service  "SysMain" -StartupType Disabled -ErrorAction SilentlyContinue
                Write-Log "Đã dọn Prefetch và tắt SuperFetch." "OK"
            }
            if ($item -match "^04") {
                $wl="EVKey|UniKey|GoogleDriveFS|SecurityHealth|WindowsDefender"
                $paths=@("HKCU:\Software\Microsoft\Windows\CurrentVersion\Run","HKLM:\Software\Microsoft\Windows\CurrentVersion\Run")
                foreach ($rp in $paths) {
                    if (Test-Path $rp) {
                        (Get-Item $rp -ErrorAction SilentlyContinue).Property | Where-Object { $_ -notmatch $wl } | ForEach-Object {
                            Remove-ItemProperty $rp $_ -Force -ErrorAction SilentlyContinue
                        }
                    }
                }
                Write-Log "Đã tắt Startup apps." "OK"
            }
            if ($item -match "^05") {
                $path="HKCU:\Software\Microsoft\Windows\CurrentVersion\BackgroundAccessApplications"
                Set-ItemProperty $path "GlobalUserDisabled" 1 -Type DWord -Force -ErrorAction SilentlyContinue
                $svcs=@("DiagTrack","dmwappushservice","XblAuthManager","XblGameSave","XboxNetApiSvc","MapsBroker","PhoneSvc","SensorService","SensrSvc","Fax","RetailDemo","WbioSrvc","PcaSvc","WalletService")
                foreach ($s in $svcs) {
                    if (Get-Service $s -ErrorAction SilentlyContinue) {
                        Stop-Service $s -Force -ErrorAction SilentlyContinue
                        Set-Service  $s -StartupType Disabled -ErrorAction SilentlyContinue
                    }
                }
                Write-Log "Đã tắt Background apps." "OK"
            }
            if ($item -match "^06") {
                Start-Process "powercfg" -ArgumentList "/setactive 8c5e7fda-e8bf-4a96-9a85-a6e23a8c635c" -Wait -WindowStyle Hidden
                Write-Log "High Performance đã kích hoạt." "OK"
            }
            if ($item -match "^07") {
                $rp="HKCU:\Software\Microsoft\Windows\CurrentVersion\Explorer\VisualEffects"
                if (-not (Test-Path $rp)) { New-Item $rp -Force | Out-Null }
                Set-ItemProperty $rp "VisualFXSetting" 3 -Type DWord -Force
                Write-Log "Đã tối ưu hiệu ứng giao diện." "OK"
            }
            if ($item -match "^08") {
                Set-ItemProperty "HKCU:\Software\Microsoft\Windows\CurrentVersion\Themes\Personalize" "EnableTransparency" 0 -Type DWord -Force
                Write-Log "Đã tắt Transparency." "OK"
            }
            if ($item -match "^09") {
                Set-ItemProperty "HKCU:\Software\Microsoft\Windows\CurrentVersion\GameDVR" "AppCaptureEnabled" 0 -Type DWord -Force -ErrorAction SilentlyContinue
                $gcs="HKCU:\System\GameConfigStore"
                if (-not (Test-Path $gcs)) { New-Item $gcs -Force | Out-Null }
                Set-ItemProperty $gcs "GameDVR_Enabled" 0 -Type DWord -Force -ErrorAction SilentlyContinue
                Write-Log "Đã tắt Game Bar và Game Mode." "OK"
            }
            if ($item -match "^10") {
                $np="HKCU:\Software\Microsoft\Windows\CurrentVersion\Notifications\Settings"
                if (-not (Test-Path $np)) { New-Item $np -Force | Out-Null }
                Set-ItemProperty $np "NOC_GlobalSetting" 0 -Type DWord -Force
                Write-Log "Đã tắt Notifications và Tips." "OK"
            }
            if ($item -match "^11") {
                $cop="HKLM:\SOFTWARE\Policies\Microsoft\Windows\WindowsCopilot"
                if (-not (Test-Path $cop)) { New-Item $cop -Force | Out-Null }
                Set-ItemProperty $cop "TurnOffWindowsCopilot" 1 -Type DWord -Force
                Stop-Process -Name "Cortana" -Force -ErrorAction SilentlyContinue
                Write-Log "Đã tắt Cortana và Copilot." "OK"
            }
            if ($item -match "^12") {
                Get-AppxPackage *Client.WebExperience* -AllUsers -ErrorAction SilentlyContinue | Remove-AppxPackage -AllUsers -ErrorAction SilentlyContinue
                Write-Log "Đã gỡ Widget." "OK"
            }

function New-DesktopShortcut($name) {
    try {
        $sh=New-Object -ComObject WScript.Shell
        $map2=@{
            "Chrome"      = "C:\Program Files\Google\Chrome\Application\chrome.exe"
            "Firefox"     = "C:\Program Files\Mozilla Firefox\firefox.exe"
            "AnyDesk"     = "C:\Program Files (x86)\AnyDesk\AnyDesk.exe"
            "UltraViewer" = "C:\Program Files (x86)\UltraViewer\UltraViewer_Desktop.exe"
            "7-Zip"       = "C:\Program Files\7-Zip\7zFM.exe"
            "WinRAR"      = "C:\Program Files\WinRAR\WinRAR.exe"
            "VLC"         = "C:\Program Files\VideoLAN\VLC\vlc.exe"
        }
        foreach ($k in $map2.Keys) {
            if ($name -match $k -and (Test-Path $map2[$k])) {
                $lnk=$sh.CreateShortcut("$([Environment]::GetFolderPath('Desktop'))\${name}.lnk")
                $lnk.TargetPath=$map2[$k]; $lnk.WorkingDirectory=Split-Path $map2[$k]; $lnk.Save(); break
            }
        }
    } catch {}
    finally { if($sh){[System.Runtime.InteropServices.Marshal]::ReleaseComObject($sh)|Out-Null} }
}

function Scan-Apps {
    Write-Log "Đang quét danh sách ứng dụng đã cài..." "INFO"
    $listUn.Items.Clear(); $global:UnTable=@{}
    $regPaths=@(
        @{Hive=[Microsoft.Win32.Registry]::LocalMachine;Path="SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall"}
        @{Hive=[Microsoft.Win32.Registry]::LocalMachine;Path="SOFTWARE\Wow6432Node\Microsoft\Windows\CurrentVersion\Uninstall"}
        @{Hive=[Microsoft.Win32.Registry]::CurrentUser; Path="SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall"}
    )
    foreach ($rp in $regPaths) {
        $key=$rp.Hive.OpenSubKey($rp.Path)
        if (-not $key) { continue }
        foreach ($sub in $key.GetSubKeyNames()) {
            $k=$key.OpenSubKey($sub); if (-not $k) { continue }
            $dn=$k.GetValue("DisplayName"); $un=$k.GetValue("UninstallString")
            $sc=$k.GetValue("SystemComponent"); $rt=$k.GetValue("ReleaseType")
            $pub=$k.GetValue("Publisher"); $ver=$k.GetValue("DisplayVersion")
            $date=$k.GetValue("InstallDate"); $sz=$k.GetValue("EstimatedSize")
            if ($dn -and $un -and ($sc -ne 1) -and ($rt -ne "Update") `
                -and $dn -notmatch "^Microsoft Visual C\+\+" `
                -and $dn -notmatch "^Microsoft \.NET") {
                $name=$dn.ToString().Trim()
                if (-not $global:UnTable.ContainsKey($name)) {
                    $global:UnTable[$name]=@{Un=$un.ToString();Publisher=$pub;Version=$ver;Date=$date;Size=$sz}
                    [void]$listUn.Items.Add($name)
                }
            }
            $k.Close()
        }
        $key.Close()
    }
    $lblUnCount.Text="Tìm thấy $($listUn.Items.Count) ứng dụng"
    Write-Log "Quét xong: $($listUn.Items.Count) ứng dụng." "OK"
}

})

        $txtUnInfo.ForeColor=$C.LogCyan
        $txtUnInfo.Text="Tên ứng dụng: $n`nNhà phát hành: $($d.Publisher)`nPhiên bản: $($d.Version)`nNgày cài: $($d.Date)`nDung lượng: $szMB MB`nLệnh gỡ: $($d.Un)"
    }
})

    $c=0; $ok=0

        try {
            $un=$d.Un
            if ($un -match '\{[A-Z0-9\-]{36}\}') {
                Start-Process msiexec.exe -ArgumentList "/x $($Matches[0]) /qn /norestart" -Wait -ErrorAction SilentlyContinue
            } elseif ($un -match '^"(.+?)"(.*)$') {
                Start-Process $Matches[1] -ArgumentList "$($Matches[2]) /S /silent /VERYSILENT /qn" -Wait -ErrorAction SilentlyContinue
            } else {
                $parts=$un -split ' ',2
                Start-Process $parts[0] -ArgumentList "$($parts[1]) /S /silent /VERYSILENT /qn" -Wait -ErrorAction SilentlyContinue
            }
            Write-Log "Đã gỡ: ${item}" "OK"; $ok++
        } catch { Write-Log "Lỗi gỡ ${item}: $($_.Exception.Message)" "ERR" }
    }
    Scan-Apps; Set-Progress 100
    Write-Log "GỠ BỎ HOÀN TẤT: $ok/$($sel.Count)" "TITLE"
})

        $cpu =(Get-CimInstance Win32_Processor -ErrorAction SilentlyContinue | Select-Object -First 1).Name
        $ram =[math]::Round($os.TotalVisibleMemorySize/1MB,1)
        $free=[math]::Round($os.FreePhysicalMemory/1MB,1)
        $disk=Get-PSDrive C | Select-Object -First 1
        $df  =[math]::Round($disk.Free/1GB,1); $du=[math]::Round($disk.Used/1GB,1)
        $lblSys.Text="  HỆ ĐIỀU HÀNH: $($os.Caption)  |  CPU: $($cpu.Trim())  |  RAM: $free / $ram GB  |  Ổ C: $df GB trống / $du GB đã dùng"
    } catch { $lblSys.Text="  Không thể thu thập thông tin hệ thống." }
    Scan-Apps
    Write-Log "WIN TOOLS Pro v26.6.25 khởi động thành công (Tối ưu 1024x768)." "TITLE"
    Write-Log "Phát triển bởi Mr Hạnh - 0938.608.602 - khanggiaphuc.com" "INFO"

        $wv=(winget --version 2>$null)
        $lblWingetCheck.Text="Winget $wv sẵn sàng."
        $lblWingetCheck.ForeColor=$C.LogGreen
    } else {
        $lblWingetCheck.Text="Winget chưa cài! Vào Microsoft Store cài App Installer."
        $lblWingetCheck.ForeColor=$C.LogRed
    }
